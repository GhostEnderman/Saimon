const audio=document.getElementById('song'), musicBtn=document.getElementById('musicBtn'), status=document.getElementById('musicStatus'); musicBtn.addEventListener('click', async () => { if (audio.paused) { status.textContent = 'Loading song...'; try { await audio.play(); musicBtn.textContent = 'Ⅱ'; status.textContent = 'Now playing'; } catch (e) { console.error('Audio error:', e); status.textContent = 'Song unavailable — check file'; } } else { audio.pause(); musicBtn.textContent = '▶'; status.textContent = 'Paused'; } }); audio.addEventListener('playing', () => { musicBtn.textContent = 'Ⅱ'; status.textContent = 'Now playing'; }); audio.addEventListener('pause', () => { musicBtn.textContent = '▶'; if (audio.currentTime > 0) status.textContent = 'Paused'; }); audio.addEventListener('error', () => { status.textContent = 'Could not load song'; });
